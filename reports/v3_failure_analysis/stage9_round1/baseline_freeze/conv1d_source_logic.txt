   312	class TemporalEncoder(nn.Module):
   313	    def __init__(self, in_features: int = 2, embedding_dim: int = 8, kernel_size: int = 3):
   314	        super().__init__()
   315	        padding = kernel_size // 2
   316	        self.conv = nn.Conv1d(
   317	            in_channels=in_features,
   318	            out_channels=embedding_dim,
   319	            kernel_size=kernel_size,
   320	            padding=padding,
   321	        )
   322	
   323	    def forward(self, x: torch.Tensor) -> torch.Tensor:
   324	        batch_size, num_nodes, time_steps, feature_dim = x.shape
   325	        x = x.reshape(batch_size * num_nodes, time_steps, feature_dim)
   326	        x = x.permute(0, 2, 1)
   327	        h = F.relu(self.conv(x))
   328	        h = torch.max(h, dim=2).values
   329	        return h.reshape(batch_size, num_nodes, -1)

   332	class TemporalGCN(nn.Module):
   333	    def __init__(self, input_features: int, temporal_dim: int = 8, gcn_hidden: int = 16, gcn_out: int = 8):
   334	        super().__init__()
   335	        self.temporal = TemporalEncoder(input_features, temporal_dim, kernel_size=3)
   336	        self.gcn1 = GCNLayer(temporal_dim, gcn_hidden)
   337	        self.gcn2 = GCNLayer(gcn_hidden, gcn_out)
   338	        self.node_head = nn.Linear(gcn_out, 1)
   339	        self.graph_head = nn.Linear(gcn_out, 1)
   340	
   341	    def forward(self, x: torch.Tensor, a_hat: torch.Tensor):
   342	        h = self.temporal(x)
   343	        h = F.relu(self.gcn1(h, a_hat))
   344	        h = F.relu(self.gcn2(h, a_hat))
   345	        node_logits = self.node_head(h).squeeze(-1)
   346	        graph_embedding = h.mean(dim=1)
   347	        graph_logits = self.graph_head(graph_embedding).squeeze(-1)
   348	        return graph_logits, node_logits
